Integration guide ↗
 JavaScript 
import Firecrawl from 'firecrawl';

const app = new Firecrawl({
  apiKey: process.env.AIHACKR_API_KEY,
  apiUrl: 'https://aihackr.com'
});

const page = await app.scrape('https://example.com', {
  formats: ['markdown']
});

console.log(page.markdown);
