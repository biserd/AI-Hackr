const response = await fetch(
  'https://aihackr.com/v2/scrape', {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${process.env.AIHACKR_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      url: 'https://example.com',
      formats: ['markdown']
    })
  }
);
console.log(await response.json());
